// 用 Promise 包装 setTimeout，模拟一个耗时任务 function zhh_step(zhh_name, zhh_ms) { return new Promise(function (resolve) { setTimeout(function () { console.log(zhh_name + " 完成"); resolve(); }, zhh_ms); }); } // async 函数里用 await 串起流程，Promise.all 实现并行 async function zhh_run() { await zhh_step("买原料", 6000); // 买原料 6s await zhh_step("取原料", 1000); // 取原料 1s await Promise.all([ // 和面、和馅 同时进行 zhh_step("和面", 3000), zhh_step("和馅", 4000) ]); await zhh_step("包饺子", 3000); // 两个都完成才能包饺子 await zhh_step("吃饺子", 2000); // 包完吃饺子 }
